const express = require('express');
const cors = require('cors');
const app = express();

app.use(express.json());
app.use(cors()); // Дозволяє сайту робити запити з будь-якого домену (зокрема з GitHub Pages)

// Тимчасовий масив у пам'яті сервера для збереження оголошень
let adsDatabase = [
    { id: 1, author: "Marcus Wright", text: "Куплю бізнес типу «24/7» у будь-якому районі штату.", time: "Щойно" }
];

// 1. Ендпоінт для отримання оголошень сайтом (GET)
app.get('/api/ads', (req, res) => {
    res.json(adsDatabase);
});

// 2. Ендпоінт для додавання оголошення (сюди стукатиме ігровий сервер або сайт) (POST)
app.post('/api/ads', (req, res) => {
    const { author, text } = req.body;
    
    if (!text) {
        return res.status(400).json({ success: false, message: "Текст оголошення не може бути порожнім" });
    }

    const newAd = {
        id: Date.now(),
        author: author || "Співробітник Weazel News",
        text: text,
        time: "Щойно"
    };

    adsDatabase.unshift(newAd); // Додаємо на початок списку
    
    // Обмежуємо історію останніх 50 оголошень
    if (adsDatabase.length > 50) {
        adsDatabase.pop();
    }

    console.log(`[Оголошення додано]: ${author} -> ${text}`);
    res.status(201).json({ success: true, ad: newAd });
});

// Запуск сервера на порту, який виділить хостинг, або 3000 локально
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
    console.log(`Weazel API server is running on port ${PORT}`);
});
